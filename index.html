Framed Drift

An idle drift RPG for PC. The player doesn't drive — they build the car, pick the track and the risk, and watch it run. Underneath is a race simulator that resolves the whole race at once, deterministic and independent of Unity.

  • Pure C# simulation assembly with no engine references — 10,000 races resolve in 0.06 s, and the tests run in dotnet without opening the editor
  • Offline progress uses the same scorer as live play: parity measured at 0.06 %
  • 135 automated tests written as the GDD's exit criteria — win rates, cash curve, presence uplift, failure rate
  • All content as commented JSON: 11 tracks, 57 parts, 12 affixes, 5 rivals — each rival a readable, counterable build
  • Procedural track generator, validated 200 of 200

EnemyBase.cs

Enemy damage response from Steam Veins — invulnerability gate, FMOD audio, health-threshold phase change, and execution state. C#, Unity.


public override void RespondTakeDamage(int typeDamage, float AditionalknockbackForceX, float AditionalknockbackForceY)
{
    if (isInvulnerable || isDead) return;
 
    GetComponentInChildren<IgnoreCollisionOnDamage>()?.TriggerIgnoreCollision();
 
    EventInstance soundInstance = RuntimeManager.CreateInstance(typeDamage == 2 ? soundShoot : soundSlash);
 
    switch (typeDamage)
    {
        case 0:
            // hit crouch melee
            break;
        case 1:
            // hit normal melee
            soundInstance.setPitch(UnityEngine.Random.Range(0.8f, 1.2f));
            soundInstance.start();
            soundInstance.release();
            break;
        case 2:
            // shoot
            soundInstance.start();
            soundInstance.release();
            break;
    }
    if (!isRained && enemyStats.hp <= enemyStats.maxHp / 2)
    {
        if (!isAttacking && !isTransitioning) { StartCoroutine(DropHat()); }
        else { shouldTransition = true; }
    }
    if (enemyStats.hp <= 0)
    {
        isDead = true;
        if (enemyStats.isExecutavel && !isExecutada)
        {
            StartCoroutine(ExecutavelPosAtk());
        }
        else if (enemyStats.isDefeated && isExecutada)
        {
            animator.SetTrigger("Derrotado");
        }
    }
}

DriftScorer.cs

Drift scoring from Framed Drift — the same scorer runs the live race and the offline one, so offline parity is structural rather than tested. Pure C#, no UnityEngine.


// Reduces a timeline to a Drift Score. Deliberately separate from the
// simulator: called on the untouched timeline it yields the offline result.
// Consumes no RNG, so scoring twice gives the same number.
public long Score(SegmentOutcome[] timeline, in Context ctx)
{
    if (timeline == null) return 0L;
 
    // Accumulate in double, floor once: a 12-million endgame summed in float
    // drops whole units on the last segment, and the same replay would score
    // differently depending on summation order.
    double sum = 0.0;
    for (int i = 0; i < timeline.Length; i++)
        sum += SegmentScore(in timeline[i], in ctx);
 
    double total = sum * ctx.TotalMultiplier;
    return total <= 0.0 ? 0L : (long)Math.Floor(total);
}
 
// Offline: the timeline as the simulator left it, no promotions.
public long ScoreWithoutPromotions(RaceResult result, in Context ctx)
{
    if (result?.Timeline == null) return 0L;
 
    double sum = 0.0;
    for (int i = 0; i < result.Timeline.Length; i++)
    {
        SegmentOutcome o = result.Timeline[i];
        o.QualityFinal = o.Quality;
        sum += SegmentScore(in o, in ctx);
    }
 
    double total = sum * ctx.TotalMultiplier;
    return total <= 0.0 ? 0L : (long)Math.Floor(total);
}
 
// uplift = (all promotions - none) / none. The balance suite measures this
// over 1,000 races and requires 0.20-0.30; it currently lands at 0.245.
public double MeasureUplift(RaceResult result, in Context ctx)
{
    long without = ScoreWithoutPromotions(result, in ctx);
    if (without <= 0L) return 0.0;
    long with = ScoreWithAllPromotions(result, in ctx);
    return (with - without) / (double)without;
}
